CUR_TTY="/dev/null"
GODOT2_OPTS="-r ${DISPLAY_WIDTH}x${DISPLAY_HEIGHT} -f"
GODOT_OPTS="--resolution ${DISPLAY_WIDTH}x${DISPLAY_HEIGHT} -f"

# No dialog pipe while a port runs; just print.
pm_message() {
    echo "$1" > "$CUR_TTY"
}

pm_platform_helper() {
    if [ -e "$PM_PIPE" ]; then
        PortMasterDialogExit
    fi
    printf ""
}

# Device-specific settings live on the spruce card.
[ -f /mnt/SDCARD/spruce/portmaster/mod.txt ] && source /mnt/SDCARD/spruce/portmaster/mod.txt
