CUR_TTY=/dev/null

export controlfolder="/mnt/SDCARD/Persistent/portmaster/PortMaster"

# Ports use $directory/ports/<game>; PORTS_DIR comes from spruce's launcher.
directory="$(dirname "${PORTS_DIR:-/mnt/SDCARD/Roms/ports}")"
export directory="${directory%/}/"

export PM_PYTHON="/mnt/SDCARD/Persistent/portmaster/bin/python3"
export PATH="/mnt/SDCARD/Persistent/portmaster/bin:$PATH"
export LD_LIBRARY_PATH="/mnt/SDCARD/Persistent/portmaster/lib:$LD_LIBRARY_PATH"

# Device-specific settings live on the spruce card.
[ -f /mnt/SDCARD/spruce/portmaster/control.txt ] && source /mnt/SDCARD/spruce/portmaster/control.txt

export ESUDO=""
export ESUDOKILL="-1" # for 351Elec and EmuELEC use "-1" (numeric one) or "-k"

export SDL_GAMECONTROLLERCONFIG_FILE="$controlfolder/gamecontrollerdb.txt"

SDLDBFILE="${SDL_GAMECONTROLLERCONFIG_FILE}"
[ -z "${SDLDBFILE}" ] && SDLDBFILE="${controlfolder}/gamecontrollerdb.txt"
SDLDBUSERFILE="${HOME}/.config/SDL-GameControllerDB/gamecontrollerdb.txt"
[ ! -f "${SDLDBUSERFILE}" ] && SDLDBUSERFILE="$SDL_GAMECONTROLLERCONFIG_FILE"

get_controls() {
  LOWRES="N"
  ANALOGSTICKS=$ANALOG_STICKS
  # spruce exports the pad map itself; keep it.
  sdl_controllerconfig="$SDL_GAMECONTROLLERCONFIG"
}

source $controlfolder/device_info.txt
source $controlfolder/funcs.txt

export GPTOKEYB2="$ESUDO env LD_PRELOAD=$controlfolder/libinterpose.${DEVICE_ARCH}.so $controlfolder/gptokeyb2 $ESUDOKILL"
export GPTOKEYB="$ESUDO $controlfolder/gptokeyb $ESUDOKILL"
